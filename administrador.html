"; return; } await cargarDiasNoLectivos(); // ===================================================== // 1. SUSPENSIONES DEL RANGO // ===================================================== const suspensiones = diasNoLectivosCache.filter( fecha => fecha >= desde && fecha <= hasta ); // ===================================================== // 12. "; } // ===================================================== // 13. " + "❌ Error al consultar asistencia" + "

" + String( error.message || error ) + "
"; } };
